cmake_minimum_required(VERSION 3.16)
project(b2w_sim2sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DDS_RUNTIME "${CMAKE_BINARY_DIR}/dds-lib")
file(MAKE_DIRECTORY "${DDS_RUNTIME}")
foreach(LIB ddsc ddscxx)
  file(CREATE_LINK "${REPO}/vendor/unitree_sdk2/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}/lib${LIB}.so"
    "${DDS_RUNTIME}/lib${LIB}.so.0" SYMBOLIC)
endforeach()
add_library(b2w_io_stall SHARED io_stall_probe.cpp)
target_link_libraries(b2w_io_stall PRIVATE dl)
set(CMAKE_BUILD_RPATH "${DDS_RUNTIME}")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/vendor/unitree_sdk2" sdk2)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
  OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED PATHS "${PYBIND_DIR}" NO_DEFAULT_PATH)
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import mujoco,pathlib; print(pathlib.Path(mujoco.__file__).parent)"
  OUTPUT_VARIABLE MUJOCO_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(GLOB MUJOCO_LIB "${MUJOCO_DIR}/libmujoco.so.*")
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(yaml-cpp REQUIRED)
pybind11_add_module(b2w_sdk2_native sdk2_transport.cpp)
target_link_libraries(b2w_sdk2_native PRIVATE unitree_sdk2)
add_executable(b2w_mujoco_bridge bridge.cpp
  "${REPO}/vendor/unitree_mujoco/simulate/src/joystick/joystick.cc")
target_include_directories(b2w_mujoco_bridge PRIVATE "${MUJOCO_DIR}/include"
  "${REPO}/vendor/unitree_mujoco/simulate/src")
target_link_libraries(b2w_mujoco_bridge PRIVATE unitree_sdk2 "${MUJOCO_LIB}" yaml-cpp Boost::program_options)
add_executable(b2w_sdk2_watchdog watchdog.cpp)
target_link_libraries(b2w_sdk2_watchdog PRIVATE unitree_sdk2 yaml-cpp)
add_executable(b2w_sdk2_pd_watchdog pd_watchdog.cpp)
target_link_libraries(b2w_sdk2_pd_watchdog PRIVATE unitree_sdk2 yaml-cpp)
add_executable(b2w_pd_fault_source pd_fault_source.cpp)
target_link_libraries(b2w_pd_fault_source PRIVATE unitree_sdk2 yaml-cpp)
add_executable(b2w_dds_fault_source fault_source.cpp)
target_link_libraries(b2w_dds_fault_source PRIVATE unitree_sdk2 yaml-cpp)
add_executable(b2w_actuator_probe actuator_probe.cpp
  "${REPO}/vendor/unitree_mujoco/simulate/src/joystick/joystick.cc")
target_include_directories(b2w_actuator_probe PRIVATE "${MUJOCO_DIR}/include"
  "${REPO}/vendor/unitree_mujoco/simulate/src")
target_link_libraries(b2w_actuator_probe PRIVATE unitree_sdk2 "${MUJOCO_LIB}" yaml-cpp Boost::program_options)
foreach(DC_TARGET b2w_mujoco_dc_bridge b2w_dc_actuator_probe b2w_mujoco_timed_bridge)
  if(DC_TARGET STREQUAL "b2w_mujoco_dc_bridge")
    set(SOURCE dc_bridge.cpp)
  elseif(DC_TARGET STREQUAL "b2w_mujoco_timed_bridge")
    set(SOURCE timed_bridge.cpp)
  else()
    set(SOURCE dc_actuator_probe.cpp)
  endif()
  add_executable(${DC_TARGET} ${SOURCE} "${REPO}/vendor/unitree_mujoco/simulate/src/joystick/joystick.cc")
  target_include_directories(${DC_TARGET} PRIVATE "${MUJOCO_DIR}/include" "${REPO}/vendor/unitree_mujoco/simulate/src")
  target_link_libraries(${DC_TARGET} PRIVATE unitree_sdk2 "${MUJOCO_LIB}" yaml-cpp Boost::program_options)
endforeach()
