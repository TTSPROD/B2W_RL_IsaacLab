"""Frozen, 19999-only state replay; no training or acceptance decision."""
from pathlib import Path
import json
import numpy as np
import yaml
from physics57_protocol import ROOT, save_json, sha256
from locomotion57_protocol import export_path

BASE = ROOT/'logs/replay57_19999_20260925'
CONFIG = ROOT/'configs/replay57_19999_execution_20260925.json'
SAVED = ROOT/'artifacts/upstream/2026-09-23_model_19999/env.yaml'
SEEDS = [7201, 7202, 7203, 7204]
DT, PHYSICS_DT, STEPS = .02, .002, 250
BODY_NAMES = ['base_link'] + [f'{leg}_{part}' for leg in ('FR','FL','RR','RL')
                              for part in ('hip','thigh','calf','foot')]


class ConfigLoader(yaml.SafeLoader):
    """Only the inert slice tag in the saved SceneEntityCfg is permitted."""


ConfigLoader.add_constructor('tag:yaml.org,2002:python/object/apply:builtins.slice',
    lambda loader, node: slice(*loader.construct_sequence(node)))


def saved_config():
    return yaml.load(SAVED.read_text(encoding='utf-8'), Loader=ConfigLoader)


def reward_terms():
    return {k:v for k,v in saved_config()['rewards'].items()
            if isinstance(v,dict) and v['weight'] != 0}


def metadata():
    return {'policy':19999, 'seeds':SEEDS, 'config_sha256':sha256(CONFIG),
            'policy_export_sha256':sha256(export_path(19999)),
            'sources':{p.name:sha256(p) for p in sorted((ROOT/'scripts').glob('replay57_*.py'))}}


def write_result(name, arrays, extra):
    path = BASE/f'{name}.npz'
    if path.exists() or path.with_suffix('.json').exists():
        raise FileExistsError(path)
    BASE.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(path, **arrays)
    save_json(path.with_suffix('.json'), {**metadata(), **extra,
        'trace_path':str(path.relative_to(ROOT)), 'trace_sha256':sha256(path)})
    print('DONE', name, flush=True)


def freeze():
    plan = ROOT/'configs/locomotion57_19999_replay_plan_20260925.json'
    hashes = {str(p.relative_to(ROOT)):sha256(p) for p in [plan, SAVED, export_path(19999),
        ROOT/'artifacts/upstream/2026-09-23_model_19999/upstream_model_19999.pt',
        ROOT/'configs/contact57_geometry_20260925.json',
        ROOT/'configs/contact57b_cooked_geometry_20260925.json']}
    save_json(CONFIG, {'schema':'replay57_19999_execution_v1', 'policy':19999,
        'seeds':SEEDS, 'terrain':'up_14x32', 'case':'forward_0.30',
        'command':[.3,0,0], 'snapshot_time_s':27., 'capture_horizon_s':45.,
        'physics_dt_s':PHYSICS_DT, 'policy_dt_s':DT, 'replay_horizon_s':5.,
        'budget':{'captures':4,'fresh_replays':12,'serialized_continuation_checks':4},
        'conditions':['isaac_nominal','mujoco_cooked_shapes','mujoco_source_shapes'],
        'contact_measurement':{'minimum_normal_force_N':5.,
            'slip':'normal-load-weighted RMS tangent velocity at physical contact points',
            'sampling':'each physics step, post step; no change to actor feedback'},
        'reward_ledger':'all nonzero saved reward terms; raw rates and dt-scaled integral; actual upstream functions',
        'qualification':'diagnostic only; no new acceptance threshold or policy promotion',
        'restrictions':['no PPO','no server','no robot','no upstream10000','no navigation feedback'],
        'input_sha256':hashes})


if __name__ == '__main__':
    freeze()
